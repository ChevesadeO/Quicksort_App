# ============================================================
#  UNIVERSIDAD TECNOLÓGICA DE PANAMÁ
#  Centro Regional de Colón
#  Laboratorio #3 – Eficiencia Algorítmica y Ordenamiento
#  Algoritmo: Quicksort  |  Lenguaje: Python 3
# ============================================================

import random   # genera números aleatorios
import time     # mide tiempo de ejecución


# ── FUNCIÓN 1: generar_lista ─────────────────────────────────
# Genera exactamente 20 números aleatorios en rango [1, 100].
# random.randint(a,b) retorna N donde a ≤ N ≤ b.
def generar_lista(cantidad=20, minimo=1, maximo=100):
    lista = []
    for _ in range(cantidad):
        lista.append(random.randint(minimo, maximo))
    return lista


# ── FUNCIÓN 2: particion ─────────────────────────────────────
# Esquema de partición Lomuto:
#   1. Último elemento = pivote.
#   2. Se recorre el sub-arreglo buscando elementos ≤ pivote.
#   3. Se intercambian acumulándolos a la izquierda.
#   4. Pivote se coloca en su posición definitiva (i+1).
def particion(lista, bajo, alto):
    pivote = lista[alto]       # elegir último como pivote
    i = bajo - 1              # índice del menor encontrado

    for j in range(bajo, alto):
        if lista[j] <= pivote:
            i += 1
            lista[i], lista[j] = lista[j], lista[i]   # swap

    # Colocar pivote en su posición final
    lista[i + 1], lista[alto] = lista[alto], lista[i + 1]
    return i + 1               # índice definitivo del pivote


# ── FUNCIÓN 3: quicksort ─────────────────────────────────────
# Divide y vencerás recursivo:
#   DIVIDIR   → particion() ubica el pivote
#   CONQUISTAR → recursión en sub-arreglos izq. y der.
#   COMBINAR  → no necesario (ordena in-place)
def quicksort(lista, bajo, alto):
    if bajo < alto:                       # caso base: 0 ó 1 elemento
        idx = particion(lista, bajo, alto)
        quicksort(lista, bajo, idx - 1)    # recursión izquierda
        quicksort(lista, idx + 1, alto)    # recursión derecha


# ── FUNCIÓN 4: mostrar_lista ─────────────────────────────────
def mostrar_lista(lista, etiqueta):
    elementos = ", ".join(str(n) for n in lista)
    print(f"  {etiqueta}: [{elementos}]")


# ── PROGRAMA PRINCIPAL ───────────────────────────────────────
def main():
    print("=" * 60)
    print("   LABORATORIO #3 – QUICKSORT  |  UTP Colón")
    print("=" * 60)

    lista_original    = generar_lista()
    lista_para_ordenar = lista_original.copy()  # copia para preservar original

    print(f"\n  Números generados: 20  |  Rango: 1 – 100\n")
    mostrar_lista(lista_original,    "Lista ORIGINAL (desordenada)")

    inicio = time.perf_counter()
    quicksort(lista_para_ordenar, 0, len(lista_para_ordenar) - 1)
    fin    = time.perf_counter()

    mostrar_lista(lista_para_ordenar, "Lista ORDENADA  (Quicksort)")
    tiempo_ms = (fin - inicio) * 1_000
    print(f"\n  Tiempo de ejecución : {tiempo_ms:.6f} ms")
    print(f"  Complejidad         : O(n log n) caso promedio")
    print("=" * 60)

if __name__ == "__main__":
    main()

1
generar_lista()
Crea una lista de 20 enteros aleatorios entre 1 y 100 usando random.randint(). Se hace una copia con .copy() antes de ordenar para preservar el arreglo original y poder mostrarlo al usuario.
2
particion(lista, bajo, alto)
El corazón del algoritmo — implementa el esquema Lomuto. Selecciona lista[alto] como pivote. Recorre con el índice j; cuando encuentra un elemento ≤ pivote, incrementa i y hace swap. Al terminar el recorrido, el pivote se intercambia a i+1, quedando en su posición definitiva. Retorna ese índice.
3
quicksort(lista, bajo, alto)
Función recursiva. Caso base: bajo >= alto (0 ó 1 elemento, ya ordenado — no hace nada). Caso recursivo: llama a particion() para ubicar el pivote, luego se llama a sí misma en el sub-arreglo izquierdo (menores) y en el derecho (mayores). Ordena in-place, sin memoria auxiliar.
4
main()
Orquesta todo: genera la lista, muestra el estado original, ejecuta quicksort() midiendo el tiempo con time.perf_counter() de alta resolución, y muestra el resultado con estadísticas de eficiencia.
